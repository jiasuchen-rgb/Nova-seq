<script>

(() => {

  const $ = (s)=>document.querySelector(s);

  const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));

  const rand=(a,b)=>a+Math.random()*(b-a);

  const randi=(a,b)=>Math.floor(rand(a,b+1));

 

  const heat = $("#heat"), htx = heat.getContext("2d");

  const cl = $("#clusters"), ctx = cl.getContext("2d");

 

  const btnToggle=$("#btnToggle"), btnStep=$("#btnStep"), btnReset=$("#btnReset");

  const speedSlider=$("#speed");

  const fcTypeSel=$("#fcType"), readLenSel=$("#readLen"), indexLenSel=$("#indexLen");

  const btnExport=$("#btnExport");

 

  const statusText=$("#statusText"), dot=$("#dot"), subtitle=$("#subtitle");

 

  const kvFC=$("#kvFC"), kvPlan=$("#kvPlan"), kvStage=$("#kvStage"), kvCycle=$("#kvCycle"), kvTile=$("#kvTile");

  const bar=$("#bar"), pct=$("#pct");

  const bLane=$("#bLane"), bTile=$("#bTile"), bQ30=$("#bQ30"), bYield=$("#bYield");

 

  const mYield=$("#mYield"), mQ30=$("#mQ30"), mPF=$("#mPF"), mErr=$("#mErr"), mInt=$("#mInt"), mAln=$("#mAln");

 

  const baseBars=$("#baseBars"), logEl=$("#log");

 

  const baseColors = {

    A:"rgba(255,107,138,.95)",

    C:"rgba(102,219,255,.95)",

    G:"rgba(103,243,182,.95)",

    T:"rgba(255,211,109,.95)",

  };

  const bases=["A","C","G","T"];

  const pick = (arr)=>arr[Math.floor(Math.random()*arr.length)];

 

  // Stages (NovaSeq-ish)

  const stages=[

    {name:"Cluster", label:"Cluster Generation", ticks:24},

    {name:"SBS", label:"Sequencing (SBS)", ticks:120},

    {name:"Basecall", label:"Base Calling", ticks:16},

    {name:"Demux", label:"Demultiplexing", ticks:12},

  ];

 

  let running=true;

  let speed=parseFloat(speedSlider.value);

 

  let stageIdx=0, stageTick=0, globalTick=0, cycle=0;

  let yieldBases=0; // bases called (sim)

  let errorRate=0.0;

 

  // tile map (2 lanes, each has grid tiles)

  let lane=1, tile=1;

  const tilesPerLane=24;      // displayed tile id range

  const gridW=12, gridH=8;    // heatmap grid

 

  // cluster points for imaging panel

  let clusters=[];

  function resetClusters(){

    clusters=[];

    const n=420;

    for(let i=0;i<n;i++){

      clusters.push({

        x:rand(12, cl.width-12),

        y:rand(18, cl.height-12),

        r:rand(1.2,2.3),

        base:pick(bases),

        glow:0

      });

    }

  }

 

  function log(msg, tone="muted"){

    const d=document.createElement("div");

    d.className=tone;

    d.textContent=`[${new Date().toLocaleTimeString()}] ${msg}`;

    logEl.appendChild(d);

    logEl.scrollTop=logEl.scrollHeight;

    while(logEl.childNodes.length>140) logEl.removeChild(logEl.firstChild);

    // keep for export

    exportRows.push({ts:Date.now(), tone, msg});

  }

 

  function currentStage(){ return stages[stageIdx]; }

  function totalTicks(){ return stages.reduce((s,x)=>s+x.ticks,0); }

 

  function planText(){

    const rl=parseInt(readLenSel.value,10);

    const il=parseInt(indexLenSel.value,10);

    return `R1 ${rl+1} · I1 ${il} · I2 ${il} · R2 ${rl+1}`;

  }

 

  function updateHeaderKVs(){

    kvFC.textContent=fcTypeSel.value;

    kvPlan.textContent=planText();

  }

 

  function updateProgress(){

    const done=stages.slice(0,stageIdx).reduce((s,x)=>s+x.ticks,0)+stageTick;

    const p=clamp(done/totalTicks(),0,1);

    bar.style.width=(p*100).toFixed(1)+"%";

    pct.textContent=Math.round(p*100)+"%";

  }

 

  function setStatus(txt, color){

    statusText.textContent=txt;

    dot.style.background=color;

    dot.style.boxShadow=`0 0 18px ${color.replace("0.95","0.55").replace("1)","0.55)")}`;

  }

 

  function enterStage(i){

    stageIdx=i; stageTick=0;

    kvStage.textContent=currentStage().name;

 

    if(currentStage().name==="Cluster"){

      log("Run started: flow cell loaded, vacuum + optics calibration…","good");

      log("Cluster generation in progress (patterned flow cell simulation).","good");

    }

    if(currentStage().name==="SBS"){

      cycle=0;

      log("SBS started: incorporation + imaging cycles streaming.","good");

    }

    if(currentStage().name==="Basecall"){

      log("Base calling started: signal normalization & base probabilities.","warn");

    }

    if(currentStage().name==="Demux"){

      log("Demultiplexing started: index matching & sample assignment.","warn");

    }

  }

 

  // Heatmap draw

  function drawHeatmap(){

    htx.clearRect(0,0,heat.width,heat.height);

 

    // frame

    htx.save();

    htx.fillStyle="rgba(0,0,0,.18)";

    htx.fillRect(0,0,heat.width,heat.height);

    htx.strokeStyle="rgba(255,255,255,.10)";

    htx.strokeRect(10,10,heat.width-20,heat.height-20);

 

    // title

    htx.fillStyle="rgba(255,255,255,.55)";

    htx.font=`12px ${getComputedStyle(document.documentElement).getPropertyValue("--mono")}`;

    htx.fillText("Tile Heatmap (PF / Q)", 18, 28);

 

    const innerX=16, innerY=42, innerW=heat.width-32, innerH=heat.height-58;

    const cellW=innerW/gridW, cellH=innerH/gridH;

 

    // generate values based on stage/cycle to feel alive

    const t = globalTick / Math.max(1,totalTicks());

    for(let y=0;y<gridH;y++){

      for(let x=0;x<gridW;x++){

        const hot = 0.55 + 0.35*Math.sin((x+y)*0.6 + t*5.2) + rand(-0.12,0.12);

        const v = clamp(hot, 0.02, 0.98);

 

        // map v -> color ramp (cyan -> purple -> green)

        // simple mix

        const r = Math.round(40 + 120*v);

        const g = Math.round(80 + 160*v);

        const b = Math.round(120 + 130*(1-v));

        htx.fillStyle = `rgba(${r},${g},${b},${0.85})`;

 

        const px = innerX + x*cellW;

        const py = innerY + y*cellH;

        htx.fillRect(px+1,py+1,cellW-2,cellH-2);

 

        // occasionally show "bad tiles" in SBS later cycles

        if(currentStage().name==="SBS" && t>0.45 && Math.random()<0.01){

          htx.fillStyle="rgba(255,107,138,.85)";

          htx.fillRect(px+1,py+1,cellW-2,cellH-2);

        }

      }

    }

 

    // lane label

    htx.fillStyle="rgba(255,255,255,.45)";

    htx.fillText(`Lane ${lane} · Tile ${String(tile).padStart(2,"0")}`, 18, heat.height-18);

 

    htx.restore();

  }

 

  function drawClusters(){

    ctx.clearRect(0,0,cl.width,cl.height);

 

    // backdrop

    ctx.fillStyle="rgba(0,0,0,.14)";

    ctx.fillRect(0,0,cl.width,cl.height);

 

    // subtle lane band

    ctx.save();

    ctx.globalAlpha=0.35;

    ctx.strokeStyle="rgba(255,255,255,.08)";

    ctx.strokeRect(10,10,cl.width-20,cl.height-20);

    ctx.fillStyle="rgba(255,255,255,.50)";

    ctx.font=`12px ${getComputedStyle(document.documentElement).getPropertyValue("--mono")}`;

    ctx.fillText("Imaging View (clusters)", 18, 28);

    ctx.restore();

 

    const st=currentStage().name;

 

    for(const c of clusters){

      let col="rgba(255,255,255,.22)";

      let rr=c.r;

 

      if(st==="Cluster"){

        rr = c.r + (stageTick*0.06);

        col="rgba(255,255,255,.26)";

      } else if(st==="SBS"){

        col=baseColors[c.base];

      } else if(st==="Basecall"){

        col="rgba(102,219,255,.35)";

      } else if(st==="Demux"){

        col="rgba(103,243,182,.30)";

      }

 

      if(st==="SBS"){

        // pulse

        if(Math.random()<0.18){ c.base=pick(bases); c.glow=1; }

        c.glow*=0.62;

      } else {

        c.glow*=0.5;

      }

 

      if(c.glow>0.02){

        ctx.beginPath();

        ctx.fillStyle = col.replace(".95","0.10");

        ctx.arc(c.x,c.y,rr+10*c.glow,0,Math.PI*2);

        ctx.fill();

      }

 

      ctx.beginPath();

      ctx.globalAlpha = (st==="SBS") ? 0.88 : 0.55;

      ctx.fillStyle=col;

      ctx.arc(c.x,c.y,rr,0,Math.PI*2);

      ctx.fill();

      ctx.globalAlpha=1;

    }

 

    // imaging sweep line during SBS

    if(st==="SBS"){

      const x = (globalTick*9) % (cl.width + 180) - 90;

      const g = ctx.createLinearGradient(x-90,0,x+90,0);

      g.addColorStop(0,"rgba(255,255,255,0)");

      g.addColorStop(0.5,"rgba(255,255,255,.14)");

      g.addColorStop(1,"rgba(255,255,255,0)");

      ctx.fillStyle=g;

      ctx.fillRect(x-90,0,180,cl.height);

    }

  }

 

  function updateTiles(){

    // rotate lane/tile for display

    lane = 1 + (Math.floor(globalTick/6) % 2);

    tile = 1 + (Math.floor(globalTick/3) % tilesPerLane);

    bLane.textContent=String(lane);

    bTile.textContent=String(tile).padStart(2,"0");

    kvTile.textContent=`L${lane}-T${String(tile).padStart(2,"0")}`;

    subtitle.textContent = `Lane ${lane} · Tile ${String(tile).padStart(2,"0")} · ${currentStage().label}`;

  }

 

  function updateMetrics(){

    const t=globalTick/Math.max(1,totalTicks());

    // yield grows mainly in SBS

    if(currentStage().name==="SBS"){

      yieldBases += randi(2_500_000, 6_500_000); // per tick

    } else if(currentStage().name==="Basecall"){

      yieldBases += randi(800_000, 2_000_000);

    }

 

    const yieldGb = yieldBases / 1e9; // "Gb"

    const q30 = clamp(92 - 10*t + rand(-1.2,1.2), 50, 95);

    const pf  = clamp(84 + 8*Math.sin(t*1.6) + rand(-1.5,1.5), 60, 98);

    errorRate = clamp(0.15 + 1.8*t + rand(-0.06,0.06), 0.05, 2.8);

    const intensity = Math.round(900 + 520*Math.sin(t*2.4) + rand(-90,90));

    const aligned = clamp(70 + 22*Math.sin(t*1.15) + rand(-2,2), 35, 99);

 

    mYield.textContent = `${yieldGb.toFixed(2)} Gb`;

    mQ30.textContent = `${q30.toFixed(1)}%`;

    mPF.textContent = `${pf.toFixed(1)}%`;

    mErr.textContent = `${errorRate.toFixed(2)}%`;

    mInt.textContent = `${intensity}`;

    mAln.textContent = `${aligned.toFixed(1)}%`;

 

    bYield.textContent = `${yieldGb.toFixed(1)} Gb`;

    bQ30.textContent = `${q30.toFixed(1)}%`;

  }

 

  function updateBaseComposition(){

    // simulate base composition drift

    const t=globalTick/Math.max(1,totalTicks());

    const a = clamp(0.25 + 0.05*Math.sin(t*4.2) + rand(-0.01,0.01), 0.10, 0.45);

    const c = clamp(0.25 + 0.05*Math.sin(t*3.6 + 1) + rand(-0.01,0.01), 0.10, 0.45);

    const g = clamp(0.25 + 0.05*Math.sin(t*3.2 + 2) + rand(-0.01,0.01), 0.10, 0.45);

    let tt = 1 - (a+c+g);

    tt = clamp(tt, 0.10, 0.45);

 

    // normalize

    const s=a+c+g+tt;

    const A=a/s, C=c/s, G=g/s, T=tt/s;

 

    const rows=[

      ["A",A],["C",C],["G",G],["T",T]

    ];

 

    baseBars.innerHTML="";

    for(const [b,v] of rows){

      const row=document.createElement("div");

      row.className="bb";

      row.innerHTML = `

        <div style="color:${baseColors[b]};font-weight:900">${b}</div>

        <div class="track"><div class="fill ${b}" style="width:${(v*100).toFixed(1)}%"></div></div>

        <div style="text-align:right">${(v*100).toFixed(1)}%</div>

      `;

      baseBars.appendChild(row);

    }

  }

 

  function maybeAdvance(){

    if(stageTick >= currentStage().ticks){

      if(stageIdx < stages.length-1){

        enterStage(stageIdx+1);

      } else {

        running=false;

        btnToggle.textContent="▶ 开始";

        setStatus("COMPLETE", "rgba(255,211,109,.95)");

        log("Run complete: BCL/FASTQ export ready (sim).","good");

      }

    }

  }

 

  function sbsCycleStep(){

    const rl=parseInt(readLenSel.value,10);

    // Increase cycle during SBS primarily; also during Basecall to show progress

    if(currentStage().name==="SBS"){

      cycle += 1;

      // clamp by plan total cycles (rough)

      const il=parseInt(indexLenSel.value,10);

      const total = (rl+1) + il + il + (rl+1);

      if(cycle > total) cycle = total;

      kvCycle.textContent = String(cycle);

    } else if(currentStage().name==="Basecall"){

      // keep cycle static, but show late progress

      kvCycle.textContent = String(cycle);

    } else {

      kvCycle.textContent = (currentStage().name==="Cluster") ? "—" : kvCycle.textContent;

    }

  }

 

  // export

  let exportRows=[];

  function exportCSV(){

    const header="timestamp,tone,message\n";

    const lines=exportRows.map(r=>{

      const msg = r.msg.replaceAll('"','""');

      return `${r.ts},${r.tone},"${msg}"`;

    }).join("\n");

    const blob=new Blob([header+lines],{type:"text/csv;charset=utf-8"});

    const url=URL.createObjectURL(blob);

    const a=document.createElement("a");

    a.href=url;

    a.download="novaseq_sim_log.csv";

    document.body.appendChild(a);

    a.click();

    a.remove();

    URL.revokeObjectURL(url);

  }

 

  // tick

  function tick(){

    if(!running) return;

 

    stageTick += 1;

    globalTick += 1;

 

    updateTiles();

    updateMetrics();

    updateBaseComposition();

    updateProgress();

 

    // occasional event messages

    const st=currentStage().name;

    if(st==="Cluster" && stageTick===8) log("Patterned flow cell loading OK. ExAmp mix stable.","good");

    if(st==="SBS" && stageTick===3) log("Imaging focus locked. Starting cycle acquisition.","good");

    if(st==="SBS" && stageTick%18===0) log(`Cycle ${cycle}: optics + chemistry stable (sim).`,"muted");

    if(st==="SBS" && Math.random()<0.02) log("Minor intensity fluctuation detected (auto-correct).","warn");

    if(st==="Basecall" && stageTick===2) log("Generating BCL files…","warn");

    if(st==="Demux" && stageTick===2) log("Matching indexes to SampleSheet entries…","warn");

 

    sbsCycleStep();

 

    drawHeatmap();

    drawClusters();

 

    maybeAdvance();

  }

 

  // loop

  let last=performance.now();

  function loop(now){

    const dt=now-last; last=now;

    const interval = 140 / speed;

    loop.acc=(loop.acc||0)+dt;

    while(loop.acc>=interval){

      loop.acc-=interval;

      tick();

      if(!running) break;

    }

    requestAnimationFrame(loop);

  }
